cmake_minimum_required(VERSION 3.2...3.27.5)

project(super_mario)

SET(CMAKE_CXX_FLAGS "-finput-charset=utf-8 -fexec-charset=utf-8")

set(SOURCES main.cpp)

add_executable(${PROJECT_NAME} ${SOURCES})
target_compile_features(${PROJECT_NAME} PRIVATE cxx_std_14)

add_subdirectory(os_controls)
add_subdirectory(controller)
add_subdirectory(levels)
add_subdirectory(model)
add_subdirectory(objects)
add_subdirectory(ui)
add_subdirectory(util)

if(GAME_TYPE STREQUAL "WindowsConsole")
	target_compile_definitions(${PROJECT_NAME} PRIVATE WINDOWS_CONSOLE)
elseif(GAME_TYPE STREQUAL "LinuxConsole")
	target_link_libraries(${PROJECT_NAME} ncurses pthread)
	target_compile_definitions(${PROJECT_NAME} PRIVATE LINUX_CONSOLE)
endif()

option(TEACHER_DIAGNOSTICS "Enable additional compiler warnings" OFF)
option(TEACHER_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(../tests ${CMAKE_CURRENT_BINARY_DIR}/tests)
endif()
foreach(game_target super_mario game_tests console_replay)
    if(TARGET ${game_target})
        if(TEACHER_DIAGNOSTICS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
            target_compile_options(${game_target} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
        if(TEACHER_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
            target_compile_options(${game_target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_libraries(${game_target} -fsanitize=address,undefined)
        endif()
    endif()
endforeach()
